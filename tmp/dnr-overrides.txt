  ?dnr_internal=1 로 한 번 접속하면 그 브라우저는 Shopify 분석·Meta 픽셀·GA4(동의 기반)에서 빠짐.
  되돌리기: ?dnr_internal=0. 표시는 localStorage 'dnr_internal'.

<script>
(function () {
	var p = new URLSearchParams(location.search).get('dnr_internal');
	var on;
	try {
		if (p === '1') localStorage.setItem('dnr_internal', '1');
		if (p === '0') localStorage.removeItem('dnr_internal');
		on = localStorage.getItem('dnr_internal') === '1';
	} catch (e) { on = p === '1'; }
	if (!on && p !== '0') return;
	function apply() {
		var cp = window.Shopify && window.Shopify.customerPrivacy;
		if (!cp || !cp.setTrackingConsent) return;
		var v = !on;
		cp.setTrackingConsent({ analytics: v, marketing: v, preferences: true, sale_of_data: v }, function () {});
	}
	function load() {
		if (window.Shopify && window.Shopify.customerPrivacy) return apply();
		if (window.Shopify && window.Shopify.loadFeatures) {
			window.Shopify.loadFeatures([{ name: 'consent-tracking-api', version: '0.1' }], function (err) { if (!err) apply(); });
		}
	}
	if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', load); else load();
	if (p !== null) {
		document.addEventListener('DOMContentLoaded', function () {
			var t = document.createElement('div');
			t.textContent = on ? 'Internal visit: this browser is now excluded from analytics.' : 'Internal flag removed: this browser is tracked again.';
			t.style.cssText = 'position:fixed;left:50%;bottom:24px;transform:translateX(-50%);z-index:99999;background:#0c1921;color:#fff;font:14px/1.4 -apple-system,sans-serif;padding:10px 16px;border-radius:8px';
			document.body.appendChild(t);
			setTimeout(function () { t.remove(); }, 4000);
		});
	}
})();
</script>
