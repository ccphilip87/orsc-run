    const m = document.createElement('canvas'); m.width = GW; m.height = GH; const mx = m.getContext('2d'); const md = mx.createImageData(GW, GH);
    for (let i=0;i<GW*GH;i++){ md.data[i*4+3] = sg[i] ? 255 : 0; } mx.putImageData(md, 0, 0);
    const c = document.createElement('canvas'); c.width = w; c.height = h; c.className = im.className; c.setAttribute('aria-hidden', 'true');
    const x = c.getContext('2d'); if (worn) x.filter = 'brightness(1.1)'; x.drawImage(im, 0, 0, w, h); x.filter = 'none';
    x.globalCompositeOperation = 'destination-in'; x.imageSmoothingEnabled = true; x.filter = 'blur(6px)'; x.drawImage(m, 0, 0, w, h);
    im.style.visibility = 'hidden'; im.after(c);
  }
  // 실루엣에서 얼마나 떨어졌는지(칸 단위). 사진 칸 바깥까지 넓게 계산해서 옷 둘레에 넉넉한 빈 종이를 만듦
  const OX = 40, OY = 40, EW = GW + OX*2, EH = GH + OY*2;
  function distField(sg){
    const d = new Float32Array(EW*EH).fill(1e6);
    for (let y=0;y<GH;y++) for (let x=0;x<GW;x++) if (sg[y*GW+x]) d[(y+OY)*EW + x+OX] = 0;
    const D1 = 1, D2 = 1.414;
    for (let y=0;y<EH;y++) for (let x=0;x<EW;x++){ const i = y*EW+x; let v = d[i];
      if (x>0) v = Math.min(v, d[i-1]+D1); if (y>0){ v = Math.min(v, d[i-EW]+D1); if (x>0) v = Math.min(v, d[i-EW-1]+D2); if (x<EW-1) v = Math.min(v, d[i-EW+1]+D2); } d[i] = v; }
    for (let y=EH-1;y>=0;y--) for (let x=EW-1;x>=0;x--){ const i = y*EW+x; let v = d[i];
      if (x<EW-1) v = Math.min(v, d[i+1]+D1); if (y<EH-1){ v = Math.min(v, d[i+EW]+D1); if (x<EW-1) v = Math.min(v, d[i+EW+1]+D2); if (x>0) v = Math.min(v, d[i+EW-1]+D2); } d[i] = v; }
    return d;
  }
  const items = [...document.querySelectorAll('.dnr-dense__item')].map(el => {
    const it = { el, pic: el.querySelector('.dnr-dense__pic'), w: 0 };
    const ims = [...el.querySelectorAll('.dnr-dense__pic img')];
    it.sils = [null, null];
    ims.forEach((i, k) => { i.loading = 'eager'; const go = () => { if (!i.naturalWidth) return; const sg = silhouette(i); if (!sg) return; it.sils[k] = distField(sg); cutout(i, sg, k === 1); }; if (i.complete) go(); else i.addEventListener('load', go); });
    return it;
  });
  function progress(){
    const vh = innerHeight;
    items.forEach(it => { const r = it.el.getBoundingClientRect(); const p = Math.min(1, Math.max(0, -r.top / (r.height - vh)));
      const w = Math.min(1, Math.max(0, (p - .3) / .3)); const ww = w*w*(3-2*w); if (Math.abs(ww - it.w) > .005){ it.w = ww; it.pic.style.setProperty('--w', ww.toFixed(3)); } });
  }
  addEventListener('scroll', progress, {passive:true});

  // 화면에 나오는 문구: 상품 실제 스펙 + 사이트에 이미 있는 카피만
  const COMMON = ['do nut run', 'just run.', 'not for show.', 'your time. your pace.', 'run for yourself.', 'i do nut run for something', 'colours from the river at dawn.', 'bukhan river', 'seams erased, strength kept'];
  function wordsFor(it){
    const d = it ? it.el.dataset : {}, line = d.line || '', col = d.colour || '', gen = d.gender || '';
    const L = line === 'air light'
      ? ['air light', 'opaque. nothing shows.', '100% polyester · 80gsm', 'lighter than a sip of water.', gen === 'men' ? '42g' : '33g', '42g · men / 33g · women']
      : ['air swift', 'mesh. air moves through.', '92% polyester · 8% spandex', 'open air, swift as breath.', 'stretch mesh'];
    return { lines: [line + ' — ' + col, ...L, ...COMMON], pops: [line, col, gen, ...L.slice(0, 4), 'do nut run', 'just run.', 'not for show.'] };
  }
  const GLYPH = 'donutr/\\|_-=+*<>~';
  let TL = [], POPS = [], lastSpawn = 0, lastPop = 0;
  function holesNow(){
    const fr = fv.getBoundingClientRect();
    return items.map(it => ({ it, r: it.pic.getBoundingClientRect(), s: (it.w > .5 && it.sils[1]) ? it.sils[1] : (it.sils[0] || it.sils[1]) }))
      .filter(h => h.r.bottom > fr.top && h.r.top < fr.bottom)
      .map(h => ({ it: h.it, x: h.r.left - fr.left, y: h.r.top - fr.top, w: h.r.width, h: h.r.height, s: h.s }));
  }
  // 옷·사람에서 GAP 안쪽이면 비워둘 자리
  function blocked(holes, cx, cy){
    for (const h of holes){ const GAP = h.w*.62, cwx = h.w/GW, cwy = h.h/GH;
      if (!h.s){ if (cx > h.x-GAP && cx < h.x+h.w+GAP && cy > h.y-GAP && cy < h.y+h.h+GAP) return true; continue; }
      const gx = Math.floor((cx-h.x)/cwx) + OX, gy = Math.floor((cy-h.y)/cwy) + OY;
      if (gx < 0 || gy < 0 || gx >= EW || gy >= EH) continue;
      if (h.s[gy*EW+gx]*cwx < GAP) return true; }
    return false;
  }
  function boxFree(holes, x, y, w, h){ for (let i=0;i<=4;i++) for (let j=0;j<=2;j++) if (blocked(holes, x + w*i/4, y + h*j/2)) return false; return x > 4 && y > 4 && x + w < FW - 4 && y + h < FH - 4; }
  function field(t){
    if (!fvis) return;
    const tt = t/1000, holes = holesNow();
    const cur = holes.length ? holes.reduce((a, b) => Math.abs(a.y + a.h/2 - FH/2) < Math.abs(b.y + b.h/2 - FH/2) ? a : b).it : null;
    const W2 = wordsFor(cur);
    fx.fillStyle = PAPER; fx.fillRect(0, 0, FW, FH); fx.textBaseline = 'top';
    // 아주 옅은 점 격자(전광판 바탕)
    fx.font = fontFor(ffs); fx.fillStyle = 'rgba(22,25,28,.09)';
    const lh = ffs*1.5;
    for (let y=8; y<FH; y+=lh) for (let x=6; x<FW; x+=fcw*2) if (!blocked(holes, x, y)) fx.fillText('.', x, y);
    // 타자 줄: 한 글자씩 찍히고, 다 찍히면 잠깐 머물다 한 번에 사라짐
    const maxLines = FW < 600 ? 11 : 18;
    if (!reduce && TL.length < maxLines && tt - lastSpawn > .2){
      for (let k=0;k<6;k++){
        const text = '> ' + W2.lines[Math.floor(Math.random()*W2.lines.length)];
        const w = text.length*fcw, x = 8 + Math.floor(Math.random()*Math.max(1, (FW - w - 16)/fcw))*fcw, y = 8 + Math.floor(Math.random()*((FH-16)/lh))*lh;
        if (TL.some(l => Math.abs(l.y - y) < lh*1.2 && x < l.x + l.w && l.x < x + w)) continue;
        if (!boxFree(holes, x, y, w, ffs)) continue;
        TL.push({ x, y, w, text, t0: tt, cps: 18 + Math.random()*22, hold: 1.2 + Math.random()*2.4 }); lastSpawn = tt; break;
      }
    }
    fx.font = fontFor(ffs);
    TL = TL.filter(l => { const n = Math.min(l.text.length, Math.floor((tt - l.t0)*l.cps)); const done = n >= l.text.length;
      if (done && !l.tEnd) l.tEnd = tt; if (l.tEnd && tt - l.tEnd > l.hold) return false;
      if (!boxFree(holes, l.x, l.y, l.w, ffs)) return false;
      fx.fillStyle = INK; fx.fillText(l.text.slice(0, n), l.x, l.y);
      if (!done || Math.floor(tt*3) % 2) { fx.fillRect(l.x + n*fcw + 1, l.y, fcw*.8, ffs*1.05); }               // 커서 █
      return true; });
    // 튀어나오는 단어: 크게, 가끔 반전(검은 바탕 흰 글자), 가끔 깨진 글자에서 맞춰지며 등장
    if (!reduce && tt - lastPop > .38 + Math.random()*.5){
      for (let k=0;k<8;k++){
        const text = W2.pops[Math.floor(Math.random()*W2.pops.length)]; if (!text) break;
        const size = Math.round(ffs * [1.8, 2.4, 3.2, 4.2][Math.floor(Math.random()*4)]);
        fx.font = fontFor(size, 700); const w = fx.measureText(text).width + size*.4, h = size*1.25;
        const x = Math.random()*(FW - w), y = Math.random()*(FH - h);
        if (!boxFree(holes, x, y, w, h)) continue;
        POPS.push({ x, y, w, h, size, text, t0: tt, life: .35 + Math.random()*.9, inv: Math.random() < .35, scr: Math.random() < .45 }); lastPop = tt; break;
      }
    }
    POPS = POPS.filter(p => { const age = tt - p.t0; if (age > p.life) return false;
      let s = p.text; if (p.scr && age < .32){ const k = Math.floor(age/.32 * s.length); s = s.slice(0, k) + s.slice(k).replace(/[^ ]/g, () => GLYPH[Math.floor(Math.random()*GLYPH.length)]); }
      fx.font = fontFor(p.size, 700);
      if (p.inv){ fx.fillStyle = INK; fx.fillRect(p.x, p.y, p.w, p.h); fx.fillStyle = PAPER; } else fx.fillStyle = INK;
      fx.fillText(s, p.x + p.size*.2, p.y + p.size*.12); return true; });
  }

  let last = 0;
  function loop(t){ if (t - last > 45){ last = t; if (visible) hero(t); field(t); } if (!reduce) requestAnimationFrame(loop); }
  function start(){ build(); fieldSize(); progress(); if (reduce){ hero(0); field(0); } else loop(0); }
  (document.fonts ? document.fonts.ready : Promise.resolve()).then(start);
  let rt, lw = innerWidth; addEventListener('resize', () => { if (innerWidth === lw) return; lw = innerWidth; clearTimeout(rt); rt = setTimeout(() => { build(); fieldSize(); }, 200); });
})();
