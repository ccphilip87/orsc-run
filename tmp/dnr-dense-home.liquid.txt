<script>
(function(){
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const INK = '#16191c', PAPER = '#f2f2f0';
  const POOL = " .,:-_~=+/\\|<>!;itrnuodTNUD";
  let RAMP = POOL.split('');
  const fontFor = fs => `600 ${fs}px "IBM Plex Mono", monospace`;
  function rank(font){
    const c = document.createElement('canvas'); c.width = c.height = 32; const x = c.getContext('2d');
    x.font = font; x.textBaseline = 'top';
    RAMP = POOL.split('').map(ch => { x.clearRect(0,0,32,32); x.fillText(ch,4,4); const d = x.getImageData(0,0,32,32).data; let s = 0; for (let i=3;i<d.length;i+=4) s += d[i]; return [ch,s]; })
      .sort((a,b) => a[1]-b[1]).map(p => p[0]);
  }

  /* ---------- 첫 화면 ---------- */
  const cv = document.getElementById('dnrArt'), ctx = cv.getContext('2d'), hint = document.getElementById('dnrHint');
  const img = new Image(); img.crossOrigin = 'anonymous';
  img.src = 'https://cdn.shopify.com/s/files/1/0829/4973/4621/files/ss26hv.webp?width=1600';
  let H, W, cw, chh, cols, rows, fs, lum, water, base, photo, mask, rim, mctx, rctx, dpr, src, visible = true;

  function build(){
    if (!img.naturalWidth) return;
    const cssW = cv.clientWidth || document.documentElement.clientWidth;
    if (!cssW) return requestAnimationFrame(build);
    dpr = Math.min(devicePixelRatio || 1, 2);
    fs = cssW < 600 ? 7 : cssW < 1000 ? 8 : 9;
    ctx.font = fontFor(fs); cw = ctx.measureText('M').width || fs*.6; chh = fs*1.02; rank(fontFor(fs));
    const iw = img.naturalWidth, ih = img.naturalHeight, aspect = cssW < 600 ? 1.25 : Math.min(ih/iw, .62);
    const sw = Math.min(iw, ih/aspect), sh = sw*aspect;
    src = [Math.max(0, Math.min(iw-sw, iw*.58 - sw/2)), Math.max(0, (ih-sh)/2), sw, sh];
    W = cssW; H = Math.round(W*aspect);
    cv.width = W*dpr; cv.height = H*dpr; cv.style.height = H + 'px';
    cols = Math.floor(W/cw); rows = Math.floor(H/chh);
    const s = document.createElement('canvas'); s.width = cols; s.height = rows;
    const sx = s.getContext('2d'); sx.drawImage(img, ...src, 0, 0, cols, rows);
    let d; try { d = sx.getImageData(0,0,cols,rows).data; } catch(e) { return; }
    const raw = new Float32Array(cols*rows); water = new Uint8Array(cols*rows); lum = new Float32Array(cols*rows);
    for (let i=0;i<cols*rows;i++){ const r=d[i*4], g=d[i*4+1], b=d[i*4+2]; raw[i] = (.3*r+.59*g+.11*b)/255; water[i] = (b-r>18 && b>90) ? 1 : 0; }
    const so = Array.from(raw).sort((a,b)=>a-b), lo = so[Math.floor(so.length*.02)], hi = so[Math.floor(so.length*.98)];
    for (let i=0;i<cols*rows;i++){ let ink = 1 - Math.min(1, Math.max(0, (raw[i]-lo)/(hi-lo||1))); ink = Math.pow(ink, 1.35); lum[i] = ink < .08 ? 0 : ink; }
    const mk = () => { const c = document.createElement('canvas'); c.width = cv.width; c.height = cv.height; return c; };
    photo = mk(); photo.getContext('2d').drawImage(img, ...src, 0, 0, photo.width, photo.height);
    mask = mk(); mctx = mask.getContext('2d'); rim = mk(); rctx = rim.getContext('2d');
    base = mk(); const bx = base.getContext('2d'); bx.scale(dpr,dpr); bx.fillStyle = PAPER; bx.fillRect(0,0,W,H);
    bx.font = fontFor(fs); bx.textBaseline = 'top'; bx.fillStyle = INK;
    for (let y=0;y<rows;y++) for (let x=0;x<cols;x++){ const i = y*cols+x; if (water[i]) continue; const k = Math.min(RAMP.length-1, Math.floor(lum[i]*RAMP.length)); if (k>0) bx.fillText(RAMP[k], x*cw, y*chh); }
  }
  function tear(px, py){
    if (!mctx) return;
    const r = (W < 600 ? 26 : 38)*dpr, X = px*dpr, Y = py*dpr;
    const blob = (c, rr, col) => { c.fillStyle = col; c.beginPath(); for (let i=0;i<=18;i++){ const a = i/18*Math.PI*2, j = rr*(.75+Math.random()*.4); const xx = X+Math.cos(a)*j, yy = Y+Math.sin(a)*j; i ? c.lineTo(xx,yy) : c.moveTo(xx,yy); } c.closePath(); c.fill(); };
    blob(rctx, r*1.18, '#fff'); blob(mctx, r, '#000'); hint.style.opacity = 0;
  }
  const at = e => { const b = cv.getBoundingClientRect(); return [e.clientX-b.left, e.clientY-b.top]; };
  let down = false;
  cv.addEventListener('pointerdown', e => { down = true; tear(...at(e)); });
  cv.addEventListener('pointermove', e => { if (down) tear(...at(e)); });
  addEventListener('pointerup', () => down = false);
  cv.addEventListener('touchmove', e => { const t = e.touches[0]; tear(...at(t)); }, {passive:true});
  new IntersectionObserver(es => { visible = es[0].isIntersecting; }).observe(cv);
  let comp;
  function hero(t){
    if (!base) return;
    ctx.setTransform(1,0,0,1,0,0); ctx.drawImage(base,0,0); ctx.scale(dpr,dpr);
    ctx.font = fontFor(fs); ctx.textBaseline = 'top'; ctx.fillStyle = INK;
    const tt = t/1000;
    for (let y=0;y<rows;y++){
      const sh = reduce ? 0 : Math.sin(tt*.6 + y*.35)*1.2 + tt*.8*(1 + (y%3)*.25);
      for (let x=0;x<cols;x++){ const i = y*cols+x; if (!water[i]) continue;
        const sxi = Math.floor(((x+sh)%cols+cols)%cols), sv = lum[y*cols+sxi] || lum[i], wob = reduce ? 0 : .1*Math.sin(tt*1.4 + x*.45 + y*.8);
        const k = Math.max(0, Math.min(RAMP.length-1, Math.floor((sv*.85+wob)*RAMP.length))); if (k>0) ctx.fillText(RAMP[k], x*cw, y*chh); }
    }
    ctx.setTransform(1,0,0,1,0,0); ctx.drawImage(rim,0,0);
    comp = comp || document.createElement('canvas'); if (comp.width !== cv.width){ comp.width = cv.width; comp.height = cv.height; }
    const c = comp.getContext('2d'); c.globalCompositeOperation = 'source-over'; c.clearRect(0,0,comp.width,comp.height); c.drawImage(photo,0,0);
    c.globalCompositeOperation = 'destination-in'; c.drawImage(mask,0,0); ctx.drawImage(comp,0,0);
    if (!reduce) for (const k of [mctx, rctx]){ k.globalCompositeOperation = 'destination-out'; k.fillStyle = 'rgba(0,0,0,.004)'; k.fillRect(0,0,mask.width,mask.height); k.globalCompositeOperation = 'source-over'; }
  }

  /* ---------- 상품 뒤 글자 결: 가운데만 비움 ---------- */
  const fv = document.getElementById('dnrField'), fx = fv.getContext('2d');
  let FW, FH, fcols, frows, ffs = 9, fcw, fvis = false;
  const WORD = 'donutrun';
  function fieldSize(){ const d = Math.min(devicePixelRatio||1, 2); FW = fv.clientWidth; FH = fv.clientHeight; fv.width = FW*d; fv.height = FH*d; fx.setTransform(d,0,0,d,0,0);
    ffs = FW < 600 ? 9 : 11; fx.font = fontFor(ffs); fcw = fx.measureText('M').width; fcols = Math.ceil(FW/fcw); frows = Math.ceil(FH/(ffs*1.1)); }
  new IntersectionObserver(es => { fvis = es[0].isIntersecting; }).observe(document.querySelector('.dnr-dense__items'));
  function field(t){
    if (!fvis) return;
    const tt = reduce ? 0 : t/1000; fx.fillStyle = PAPER; fx.fillRect(0,0,FW,FH); fx.font = fontFor(ffs); fx.textBaseline = 'top';
    const cx = FW/2, cy = FH*.46, rx = Math.min(FW*.40, 260), ry = Math.min(FH*.36, 330);
    for (let y=0;y<frows;y++) for (let x=0;x<fcols;x++){
      const px = x*fcw, py = y*ffs*1.1;
      const e = Math.hypot((px-cx)/rx, (py-cy)/ry); if (e < 1) continue;
      const n = .5 + .5*Math.sin(x*.16 + tt*.5) * Math.cos(y*.11 - tt*.35);   // 바람 결
      const a = Math.min(1, (e-1)*1.6) * (.12 + .45*n);
      if (a < .05) continue;
      fx.fillStyle = `rgba(22,25,28,${a.toFixed(2)})`;
      fx.fillText(WORD[(x + y*3 + Math.floor(tt*.8)) % WORD.length], px, py);
    }
  }

  // 한 장씩 나타나기
  const io = new IntersectionObserver(es => es.forEach(e => e.target.classList.toggle('is-in', e.isIntersecting)), {threshold:.55});
  document.querySelectorAll('.dnr-dense__item').forEach(el => io.observe(el));

  let last = 0;
  function loop(t){ if (t - last > 45){ last = t; if (visible) hero(t); field(t); } if (!reduce) requestAnimationFrame(loop); }
  function start(){ build(); fieldSize(); loop(0); if (reduce){ hero(0); field(0); } }
  (document.fonts ? document.fonts.ready : Promise.resolve()).then(() => { if (img.complete && img.naturalWidth) start(); else img.onload = start; });
  let rt; addEventListener('resize', () => { clearTimeout(rt); rt = setTimeout(() => { build(); fieldSize(); }, 200); });
})();
</script>
