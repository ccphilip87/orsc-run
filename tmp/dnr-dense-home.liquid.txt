    return lerp(lerp(hash(xi,yi), hash(xi+1,yi), sm(xf)), lerp(hash(xi,yi+1), hash(xi+1,yi+1), sm(xf)), sm(yf)); }

  // makeRunnerFrames(rowsTall) → [{w,h,cells:[ [c,r,ch], ... ]}]
  function makeRunnerFrames(rowsTall, cellAspect, nFrames){

    const S = 4;                                   // 칸 하나를 S×S로 쪼개 그려서 덮인 비율로 판단

        cells.push([cc, rr, g]);

      frames.push({ w: Wc, h: rowsTall, cells });
    }

  }

  /* ---------- 첫 화면: 찢어진 종이 + 강가·흐르는 강·윤슬 ---------- */
  const cv = document.getElementById('dnrArt'), ctx = cv.getContext('2d');

  // 칸 하나의 글자: 언덕(경사 방향 / \ | _), 갈대(| i l !), 앞쪽 풀(, ' ; " /)
  function glyph(u, v, c, r, rc, rh, dv){
    const R = ridge(u);
    if (v < R) return (hash(c, r) > .992) ? '.' : '';                                        // 하늘: 거의 빈 종이

      if (toRiver < .035) return 'Ii|l!'[Math.floor(hash(c,r)*5)];                           // 물가 갈대
      const e = .004, h0 = hill(u, v), hx = hill(u+e, v) - h0, hy = hill(u, v+e) - h0;
      const shade = .35 + .65*Math.min(1, (v - R)/.25);
      if (hash(c, r+7) > shade*.95 + .1) return '';

        if (TL.some(l => Math.abs(l.y - y) < lh*1.2 && x < l.x + l.w && l.x < x + w)) continue;
        if (!boxFree(holes, x, y, w, ffs)) continue;
        TL.push({ x, y, w, text, t0: tt, cps: 18 + Math.random()*22, hold: 1.2 + Math.random()*2.4 }); lastSpawn = tt; break;

      return true; });
    // 튀어나오는 단어: 크게, 가끔 반전(검은 바탕 흰 글자), 가끔 깨진 글자에서 맞춰지며 등장
    if (!reduce && tt - lastPop > .38 + Math.random()*.5){
      for (let k=0;k<8;k++){
        const text = W2.pops[Math.floor(Math.random()*W2.pops.length)]; if (!text) break;

        fx.font = fontFor(size, 700); const w = fx.measureText(text).width + size*.4, h = size*1.25;

      if (p.inv){ fx.fillStyle = INK; fx.fillRect(p.x, p.y, p.w, p.h); fx.fillStyle = PAPER; } else fx.fillStyle = INK;
      fx.fillText(s, p.x + p.size*.2, p.y + p.size*.12); return true; });
  }
