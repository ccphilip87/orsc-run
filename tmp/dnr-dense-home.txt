  const fbm = (x, y) => (vnoise(x,y)*.55 + vnoise(x*2.1,y*2.1)*.3 + vnoise(x*4.3,y*4.3)*.15);

     화면 (u,v) = 참고 사진(트랙 곡선부, 1344x752) 기준 가로·세로 0..1. 카메라 높이 10, 초점거리 F(가로폭 단위), 지평선 VH(화면 위 바깥).
     지면 원 중심 (X0,Z0), 안쪽 경계 반지름 R0, 레인 폭 LW, 레인 6개. 값은 참고 사진의 레인 선 위 점 18개로 맞춤. */
  const cv = document.getElementById('dnrArt'), ctx = cv.getContext('2d');
  let W, H, dpr, fs, cw, ch, cols, rows, photo, paper, mask, rim, mctx, rctx, tearAt, visible = true, heroDirty = true;
  const A = 1344/752, CAM = { F: 2.5851, VH: -.4833, X0: 29.9415, Z0: 55.7365, R0: 32.336, LW: 3.5333, N: 6 };
  let uOff = 0, uSpan = 1;                         // 화면 x → 참고 사진 u (모바일은 곡선 꼭짓점이 들어오는 세로 조각)
  // 화면 점 → 안쪽 경계에서 잰 레인 단위 거리 k (0 = 안쪽 잔디 경계, 1~6 = 레인 선), 먼 정도 Z
  function laneAt(x, y){
    const u = uOff + x/W*uSpan, v = y/H, dy = (v - CAM.VH)/A, Z = CAM.F*10/dy, X = (u - .5)*Z/CAM.F;
    return [(Math.hypot(X - CAM.X0, Z - CAM.Z0) - CAM.R0)/CAM.LW, Z];
  }
  function paintTrackPhoto(wx, s){
    // 사진 쪽은 픽셀마다 직접 칠함(먼 쪽 가는 선도 매끈하게). 한 줄씩 레인 값을 구해 옆·아랫줄과 비교 → 픽셀 하나가 레인 단위로 얼마인지
    const PW = Math.round(W*s), PH = Math.round(H*s), img = wx.createImageData(PW, PH), D = img.data;
    const RUB = [196, 73, 47], GRASS = [122, 138, 70], LINE = [243, 238, 226], half = .032;   // 선 반폭(레인 폭 대비)
    let cur = new Float32Array(PW + 1), nxt = new Float32Array(PW + 1);
    const row = (arr, py) => { const y = py/s; for (let px=0; px<=PW; px++) arr[px] = laneAt(px/s, y)[0]; };
    row(cur, 0);
    for (let py=0; py<PH; py++){
      row(nxt, py + 1);
      for (let px=0; px<PW; px++){
        const k = cur[px], pix = Math.max(Math.abs(cur[px+1] - k), Math.abs(nxt[px] - k), 1e-4);
        const onTrack = k > -.02 && k < CAM.N + .45, c = onTrack ? RUB : GRASS;
        const n = hash(px, py), g = fbm(px*.08, py*.03);
        let b = onTrack ? (n - .5)*26 : (g - .5)*40 + (n - .5)*18;
        b += (fbm(px/s*.004, py/s*.004) - .5)*14;                          // 큰 얼룩(사진 같은 밝기 흔들림)
        let r = c[0] + b, gg = c[1] + b, bl = c[2] + b*.8;
        const ln = Math.round(k);
        if (ln >= 0 && ln <= CAM.N){ const cov = Math.max(0, Math.min(1, (half - Math.abs(k - ln))/pix + .5));
          if (cov > 0){ const t = (n - .5)*10; r = lerp(r, LINE[0] + t, cov); gg = lerp(gg, LINE[1] + t, cov); bl = lerp(bl, LINE[2] + t, cov); } }
        const i = (py*PW + px)*4; D[i] = r; D[i+1] = gg; D[i+2] = bl; D[i+3] = 255;
      }
      const t = cur; cur = nxt; nxt = t;
    }
    wx.putImageData(img, 0, 0);
  }
  // 종이 쪽 한 칸: 레인 선은 화면 접선 방향대로 - \ | /, 레인 바닥은 듬성듬성, 잔디는 , ' ; ". 먼 곳일수록 글자가 성김
  function trackGlyph(x, y, c, r){
    const cx = x + cw/2, cy = y + ch/2;
    const [k, Z] = laneAt(cx, cy), [kx] = laneAt(cx + 1, cy), [ky] = laneAt(cx, cy + 1);
    const gx = kx - k, gy = ky - k, cellK = .5*(Math.abs(gx)*cw + Math.abs(gy)*ch), far = Math.min(1, 40/Z);
    if (k < -.02){ if (hash(c, r+3) > .06 + .22*far) return ''; return ",';\"/\\|.,"[Math.floor(hash(c+1, r)*9)]; }       // 안쪽 잔디
    if (k > CAM.N + .45){ if (hash(c, r+5) > .06 + .26*far) return ''; return ",'.`;"[Math.floor(hash(c+2, r)*5)]; }   // 바깥 잔디
    const ln = Math.round(k);
    if (ln >= 0 && ln <= CAM.N && Math.abs(k - ln) < Math.max(.035, cellK*.9)){
      let a = Math.atan2(gx, -gy) * 180/Math.PI; a = ((a % 180) + 180) % 180;
      if (a < 22.5 || a >= 157.5) return '-'; if (a < 67.5) return '\\'; if (a < 112.5) return '|'; return '/';
    }
    if (hash(c, r+9) > 1 - .10*far) return '.:'[Math.floor(hash(c+4, r)*2)];                                          // 레인 바닥
    return '';
  }
  // 포인트 색: 레인 선 글자의 일부만 빨강(트랙색), 아주 드물게 빨강 → 전체 글자의 5% 미만
  const accent = (g, a, b) => { const h = hash(a*7 + 3, b*13 + 5); if ('-/\\|'.includes(g) && h > .955) return RED; if (h < .012) return RED; return INK; };

    let wx; [photo, wx] = mk(Math.min(dpr, 1.5)); paintTrackPhoto(wx, photo.width/W);

    let px; [paper, px] = mk(dpr); px.scale(dpr, dpr);

    }

    let mx, rx; [mask, mx] = mk(dpr); [rim, rx] = mk(dpr); mx.scale(dpr, dpr); rx.scale(dpr, dpr); mctx = mx; rctx = rx;

  }
  function paintTear(){

    mctx.clearRect(0,0,W,H); rctx.clearRect(0,0,W,H);

    for (let i=0;i<H*1.2;i++){ const y = Math.random()*H, x0 = tearAt(y) - 4 - Math.random()*12; rctx.strokeStyle = `rgba(255,255,255,${(.4+Math.random()*.5).toFixed(2)})`; rctx.lineWidth = .6;
      rctx.beginPath(); rctx.moveTo(x0, y); rctx.lineTo(x0 - 2 - Math.random()*5, y + (Math.random()-.5)*4); rctx.stroke(); }
  }
  // 문지르면 종이가 찢겨 트랙이 드러남 (콜라주 조각은 클릭을 통과시킴)
  function tear(px, py){

  }

  }

  /* ---------- 콜라주: 조각이 다 준비되면 차례로 붙고, 데스크톱에서는 마우스를 따라 깊이별로 살짝 움직임 ---------- */

  Promise.all(pcs.map(im => im.decode ? im.decode().catch(() => {}) : Promise.resolve())).then(() => col.classList.add('go'));
  // 따라 움직임: 데스크톱은 마우스, 휴대폰은 문지르는 손가락 위치. 깊이(--d)가 클수록 많이 움직임
  let modelBusy = false;                                              // 모델이 클릭으로 흔들리는 동안 다른 조각은 멈춤
  const setPar = (mx, my) => { if (!modelBusy) pcs.forEach(p => { const d = +p.style.getPropertyValue('--d') || 0; p.style.setProperty('--px', (-mx*18*d).toFixed(1) + 'px'); p.style.setProperty('--py', (-my*12*d).toFixed(1) + 'px'); }); };
  const parAt = (x, y) => { const b = stageEl.getBoundingClientRect(); setPar((x - b.left)/b.width - .5, (y - b.top)/b.height - .5); };
  if (!reduce){

      stageEl.addEventListener('pointerleave', () => setPar(0, 0));

    }
  }
  /* 가만히 있을 때: 조각마다 다른 주기로 1~3px·1° 안쪽에서 떠다님(모델은 거의 그대로).
     4~6초마다 모델이 아닌 조각 하나가 바람 맞은 듯 들렸다 내려앉음.
     스크롤: 뒤쪽 조각일수록 늦게 따라와 깊이가 생김. 맨 앞 슬로건은 그대로(히어로 바닥에 붙어 있게)
     히어로를 클릭(데스크톱)·탭(휴대폰)하면 모델이 같은 식으로 흔들리고, 그동안 다른 조각은 떠다님·마우스 따라감을 멈춤 */
  if (!reduce){
    const FL = pcs.map(p => {
      const person = p.classList.contains('dnr-pc--person'), z = +p.style.getPropertyValue('--z') || 0, k = person ? .3 : 1, sg = () => Math.random() < .5 ? -1 : 1;
      return { p, person, front: z >= 7, ax: (1 + Math.random()*2)*k*sg(), ay: (1.5 + Math.random()*1.5)*k*sg(), ar: (.3 + Math.random()*.6)*(person ? .25 : 1)*sg(),
        tx: 5 + Math.random()*4, ty: 6 + Math.random()*4, tr: 7 + Math.random()*4, ph: Math.random()*6.283,
        sd: z >= 7 ? 0 : person ? .035 : .06 + (4 - Math.min(z, 4))*.012 };                  // 스크롤 깊이(히어로 높이 대비)
    });
    const model = FL.find(f => f.person);
    let heroOn = true, fl = null, nextFl = 2.5, tf = 0, tPrev = 0;          // tf = 떠다님 시계(모델이 흔들리는 동안 멈춰서 다시 움직일 때 튀지 않음)
    stageEl.addEventListener('click', () => { if (!model) return; const t = performance.now()/1000; fl = { f: model, t0: t, solo: true }; nextFl = t + .6 + 4 + Math.random()*2; });
    new IntersectionObserver(es => { heroOn = es[0].isIntersecting; }).observe(stageEl);
    const anim = now => {
      requestAnimationFrame(anim); const t = now/1000, dt = tPrev ? Math.min(.1, t - tPrev) : 0; tPrev = t; if (!heroOn) return;

      const hh = stageEl.offsetHeight, s = Math.max(0, Math.min(1, -stageEl.getBoundingClientRect().top / hh));

      for (const f of FL){

        const wy = Math.sin(tf*6.283/f.ty + f.ph*1.7), fy = f.front ? Math.abs(f.ay)*(.5 + .5*wy) : f.ay*wy;
        let x = f.ax*Math.sin(tf*6.283/f.tx + f.ph), y = fy + f.sd*hh*s, r = f.ar*Math.sin(tf*6.283/f.tr + f.ph*.6), sc = 1;
        if (fl && fl.f === f){ const u = (t - fl.t0)/(fl.solo ? .6 : 1.2);
          if (u >= 1) fl = null; else { const e = Math.sin(Math.PI*u); r += Math.sin(u*6.283*(fl.solo ? 1.25 : 2.5))*(1 - u)*1.8; if (!f.front) y -= 4*e; sc = 1 + .025*e; } }
        f.p.style.transform = `translate(${x.toFixed(2)}px, ${y.toFixed(2)}px) rotate(${r.toFixed(3)}deg) scale(${sc.toFixed(4)})`;

  /* ---------- 상품 영역: 터미널 타자 + 툭 튀어나오는 단어 ---------- */
  const fv = document.getElementById('dnrField'), fx = fv.getContext('2d');
  let FW, FH, ffs = 11, fcw, fvis = false;
  const GW = 48, GH = 60;
  function fieldSize(){ const d = Math.min(devicePixelRatio||1, 2); FW = fv.clientWidth; FH = fv.clientHeight; fv.width = FW*d; fv.height = FH*d; fx.setTransform(d,0,0,d,0,0);
    ffs = FW < 600 ? 11 : 13; fx.font = fontFor(ffs); fcw = fx.measureText('M').width; }
  new IntersectionObserver(es => { fvis = es[0].isIntersecting; }).observe(document.querySelector('.dnr-dense__items'));
  // 사진 바탕색(모서리 평균)과 다른 칸 = 옷·사람. 한 칸씩 부풀려서 글자가 살짝 떨어지게
  function silhouette(im){
    const c = document.createElement('canvas'); c.width = GW; c.height = GH; const x = c.getContext('2d');
    x.drawImage(im, 0, 0, GW, GH); let d; try { d = x.getImageData(0,0,GW,GH).data; } catch(e) { return null; }
    const px = i => [d[i*4], d[i*4+1], d[i*4+2]];
    const cs = [0, GW-1, (GH-1)*GW, GH*GW-1].map(px), bg = [0,1,2].map(k => cs.reduce((a,c) => a + c[k], 0)/4);
    // 스튜디오 바탕은 밝고 색이 거의 없음(가운데가 더 밝은 조명 얼룩 있음) → 색이 있거나 바탕보다 확실히 어두우면 옷·사람
    const bgL = (bg[0]+bg[1]+bg[2])/3, g = new Uint8Array(GW*GH);
    for (let i=0;i<GW*GH;i++){ const [r,gg,b] = px(i); const sat = Math.max(r,gg,b) - Math.min(r,gg,b), L = (r+gg+b)/3; if (sat > 16 || L < bgL - 40) g[i] = 1; }
    const seen = new Uint8Array(GW*GH), q = [];
    for (let x2=0;x2<GW;x2++){ q.push(x2, (GH-1)*GW+x2); } for (let y2=0;y2<GH;y2++){ q.push(y2*GW, y2*GW+GW-1); }
    while (q.length){ const i = q.pop(); if (seen[i] || g[i]) continue; seen[i] = 1; const x2 = i%GW, y2 = (i/GW)|0;
      if (x2>0) q.push(i-1); if (x2<GW-1) q.push(i+1); if (y2>0) q.push(i-GW); if (y2<GH-1) q.push(i+GW); }
    for (let i=0;i<GW*GH;i++) if (!seen[i]) g[i] = 1;
    const out = new Uint8Array(GW*GH), R = 2;
    for (let y=0;y<GH;y++) for (let xx=0;xx<GW;xx++){ let on = 0; for (let dy=-R;dy<=R && !on;dy++) for (let dx=-R;dx<=R;dx++){ const X = xx+dx, Y = y+dy; if (X>=0 && Y>=0 && X<GW && Y<GH && g[Y*GW+X]){ on = 1; break; } } out[y*GW+xx] = on; }
    return out;
  }
  // 실루엣 모양으로 사진을 오려 캔버스로 바꿔 끼움 (사진 바탕이 네모로 글자를 가리지 않게)
  function cutout(im, sg, worn){
    const w = Math.min(im.naturalWidth, 900), h = Math.round(w * im.naturalHeight / im.naturalWidth);
    const m = document.createElement('canvas'); m.width = GW; m.height = GH; const mx = m.getContext('2d'); const md = mx.createImageData(GW, GH);
    for (let i=0;i<GW*GH;i++){ md.data[i*4+3] = sg[i] ? 255 : 0; } mx.putImageData(md, 0, 0);

    // 사진 바탕(회색 스튜디오)에 가까운 픽셀일수록 종이색으로 바꿈 → 오려낸 둘레에 회색 테두리가 안 남음
    try {
      const id = x.getImageData(0, 0, w, h), D = id.data, P = [242, 242, 240];
      const k = (xx, yy) => { const i = (yy*w + xx)*4; return [D[i], D[i+1], D[i+2]]; };
      const cs = [k(2,2), k(w-3,2), k(2,h-3), k(w-3,h-3), k(w>>1,2)], bg = [0,1,2].map(j => cs.reduce((a, c2) => a + c2[j], 0)/cs.length);
      const bgL = (bg[0]+bg[1]+bg[2])/3, cl = v => v < 0 ? 0 : v > 1 ? 1 : v;
      for (let i=0;i<D.length;i+=4){
        const r = D[i], g = D[i+1], b = D[i+2], mx = Math.max(r,g,b), mn = Math.min(r,g,b), L = (r+g+b)/3;
        const t = cl(1 - (mx - mn - 8)/18) * cl(1 - (bgL - L - 4)/26);       // 색이 없고 바탕만큼 밝을수록 1
        if (t <= 0) continue;
        D[i] = r + (P[0]-r)*t; D[i+1] = g + (P[1]-g)*t; D[i+2] = b + (P[2]-b)*t;
      }
      x.putImageData(id, 0, 0);
    } catch (e) {}
    x.globalCompositeOperation = 'destination-in'; x.imageSmoothingEnabled = true; x.filter = 'blur(6px)'; x.drawImage(m, 0, 0, w, h);
    im.style.visibility = 'hidden'; im.after(c);
  }
  // 실루엣에서 얼마나 떨어졌는지(칸 단위). 사진 칸 바깥까지 넓게 계산해서 옷 둘레에 넉넉한 빈 종이를 만듦
  const OX = 40, OY = 40, EW = GW + OX*2, EH = GH + OY*2;
  function distField(sg){
    const d = new Float32Array(EW*EH).fill(1e6);
    for (let y=0;y<GH;y++) for (let x=0;x<GW;x++) if (sg[y*GW+x]) d[(y+OY)*EW + x+OX] = 0;
    const D1 = 1, D2 = 1.414;
    for (let y=0;y<EH;y++) for (let x=0;x<EW;x++){ const i = y*EW+x; let v = d[i];
      if (x>0) v = Math.min(v, d[i-1]+D1); if (y>0){ v = Math.min(v, d[i-EW]+D1); if (x>0) v = Math.min(v, d[i-EW-1]+D2); if (x<EW-1) v = Math.min(v, d[i-EW+1]+D2); } d[i] = v; }
    for (let y=EH-1;y>=0;y--) for (let x=EW-1;x>=0;x--){ const i = y*EW+x; let v = d[i];
      if (x<EW-1) v = Math.min(v, d[i+1]+D1); if (y<EH-1){ v = Math.min(v, d[i+EW]+D1); if (x<EW-1) v = Math.min(v, d[i+EW+1]+D2); if (x>0) v = Math.min(v, d[i+EW-1]+D2); } d[i] = v; }
    return d;
  }
  const items = [...document.querySelectorAll('.dnr-dense__item')].map(el => {
    const it = { el, pic: el.querySelector('.dnr-dense__pic'), obj: el.querySelector('.dnr-dense__obj'), desc: el.querySelector('.dnr-dense__desc'), w: 0, m: -1, shown: -1 };
    it.lines = it.desc ? [...it.desc.querySelectorAll('p')] : []; it.total = it.lines.reduce((a, p) => a + p.dataset.t.length, 0);
    it.lab = [...el.querySelectorAll('.dnr-dense__label .tw')]; it.lOn = false;
    const ims = [...el.querySelectorAll('.dnr-dense__pic img')];
    it.sils = [null, null];
    ims.forEach((i, k) => { i.loading = 'eager'; const go = () => { if (!i.naturalWidth) return; const sg = silhouette(i); if (!sg) return; it.sils[k] = distField(sg); cutout(i, sg, k === 1); }; if (i.complete) go(); else i.addEventListener('load', go); });
    return it;
  });
  function progress(){
    const vh = innerHeight;
    items.forEach(it => { const r = it.el.getBoundingClientRect(); const p = Math.min(1, Math.max(0, -r.top / (r.height - vh)));
      const ease = v => { v = Math.min(1, Math.max(0, v)); return v*v*(3-2*v); };
      // 0~.18 가운데 정면 → .18~.36 왼쪽으로 → .36~.8 설명 타자 → 착용샷은 .45~.7
      const m = ease((p - .18) / .18), ww = ease((p - .45) / .25), q = Math.min(1, Math.max(0, (p - .36) / .44));
      if (Math.abs(ww - it.w) > .005){ it.w = ww; it.pic.style.setProperty('--w', ww.toFixed(3)); }
      if (Math.abs(m - it.m) > .003){ it.m = m; it.obj && it.obj.style.setProperty('--m', m.toFixed(3)); it.desc && it.desc.style.setProperty('--d', Math.min(1, m*1.4).toFixed(3)); }
      if (it.desc){ const n = Math.round(q * it.total); if (n !== it.shown){ it.shown = n; type(it, n); } it.desc.classList.toggle('is-on', q > .9); }
      // 이름표·가격: 무대가 화면 가운데에 들어오면 타자로 찍히고, 화면 밖으로 나가면 지워져서 다시 올 때 또 찍힘
      if (it.lab.length){ const on = r.top < vh*.45 && r.bottom > vh*.55, near = r.top < vh && r.bottom > 0;
        if (on && !it.lOn){ it.lOn = true; labelRun(it); } else if (!near && it.lOn){ it.lOn = false; clearInterval(it.lt); labelSet(it, 0); } } });
  }
  // 스크롤 진행만큼 설명이 한 글자씩 찍힘 (위로 올리면 지워짐), 마지막 글자 뒤에 커서
  function type(it, n){
    let left = n, curSet = false;
    for (const p of it.lines){
      const t = p.dataset.t, k = Math.max(0, Math.min(t.length, left)); left -= t.length;
      const txt = t.slice(0, k);

      else p.textContent = txt;
      if (!curSet && (k < t.length || p === it.lines[it.lines.length-1])){ if (k > 0 || n === 0) { const c = document.createElement('span'); c.className = 'cur'; p.appendChild(c); } curSet = true; }
    }
  }

  /* 사진 아래 이름표·가격: 글자를 지우지 않고 왼쪽부터 한 글자 폭씩 드러냄(가격은 현지 통화로 바뀌어도 그대로 따라감).
     드러나는 끝에 빨간 커서, 다 찍히면 잠깐 깜빡이고 사라짐 */
  const labLen = tw => Math.max(1, tw.textContent.trim().length);
  function labelSet(it, n){                                            // n = 지금까지 찍힌 글자 수(두 줄 합)
    let left = n, curSet = false;

  }
  // 옷·사람에서 GAP 안쪽이면 비워둘 자리
  function blocked(holes, cx, cy){
    for (const r of PANELS) if (cx > r.x - 18 && cx < r.x + r.w + 18 && cy > r.y - 18 && cy < r.y + r.h + 18) return true;
    for (const h of holes){ const GAP = h.w*.62, cwx = h.w/GW, cwy = h.h/GH;
      if (!h.s){ if (cx > h.x-GAP && cx < h.x+h.w+GAP && cy > h.y-GAP && cy < h.y+h.h+GAP) return true; continue; }
      const gx = Math.floor((cx-h.x)/cwx) + OX, gy = Math.floor((cy-h.y)/cwy) + OY;
      if (gx < 0 || gy < 0 || gx >= EW || gy >= EH) continue;
      if (h.s[gy*EW+gx]*cwx < GAP) return true; }
    return false;
  }
  function boxFree(holes, x, y, w, h){ for (let i=0;i<=4;i++) for (let j=0;j<=2;j++) if (blocked(holes, x + w*i/4, y + h*j/2)) return false; return x > 4 && y > 4 && x + w < FW - 4 && y + h < FH - 4; }
  function field(t){

    for (let y=8; y<FH; y+=lh) for (let x=6; x<FW; x+=fcw*2) if (!blocked(holes, x, y)) fx.fillText('.', x, y);
  }

  let last = 0;
  function loop(t){ if (t - last > 45){ last = t; if (visible && heroDirty){ heroDirty = false; hero(); } field(t); } if (!reduce) requestAnimationFrame(loop); }
